cmake_minimum_required(VERSION 3.8)
project(m1_local_fast2d)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(m1_fast_planner REQUIRED)
find_package(m1_scope_msgs REQUIRED)
find_package(m1_scope_risk REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/TimedTrajectoryPoint.msg"
  "msg/TimedTrajectory.msg"
  "msg/LocalTrajectoryCandidate.msg"
  "msg/LocalTrajectoryCandidateArray.msg"
  "msg/CandidateRiskScore.msg"
  "msg/CandidateRiskScoreArray.msg"
  "msg/SelectedLocalTrajectory.msg"
  "msg/SelectionCandidateDiagnostic.msg"
  "msg/SelectionDiagnosticArray.msg"
  DEPENDENCIES builtin_interfaces std_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_library(local_fast2d_plugin SHARED src/hybrid_controller.cpp src/local_fast2d_core.cpp src/candidate_risk_scorer.cpp src/candidate_selector.cpp src/timed_reference_retimer.cpp src/restricted_mppi_reference.cpp src/restricted_mppi_phase_tracker.cpp src/restricted_mppi_policy.cpp src/restricted_mppi_adapter.cpp)
# rosidl reserves the package-name target; retain the historical installed
# libm1_local_fast2d.so required by the plugin XML.
set_target_properties(local_fast2d_plugin PROPERTIES OUTPUT_NAME ${PROJECT_NAME})
target_link_libraries(local_fast2d_plugin "${cpp_typesupport_target}")
target_compile_features(local_fast2d_plugin PUBLIC cxx_std_17)
target_compile_options(local_fast2d_plugin PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(local_fast2d_plugin PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(local_fast2d_plugin ament_index_cpp diagnostic_msgs m1_fast_planner m1_scope_msgs m1_scope_risk nav2_core nav2_costmap_2d nav2_mppi_controller nav2_msgs nav_msgs pluginlib rclcpp std_msgs tf2 tf2_geometry_msgs)
add_executable(m1_local_fast2d_causality_harness src/causality_harness.cpp)
target_link_libraries(m1_local_fast2d_causality_harness local_fast2d_plugin)
ament_target_dependencies(m1_local_fast2d_causality_harness diagnostic_msgs nav2_msgs nav_msgs rclcpp std_msgs tf2)
pluginlib_export_plugin_description_file(nav2_core hybrid_controller_plugin.xml)
install(TARGETS local_fast2d_plugin m1_local_fast2d_causality_harness LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES hybrid_controller_plugin.xml DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_timed_trajectory test/test_timed_trajectory.cpp)
  if(TARGET test_timed_trajectory)
    target_link_libraries(test_timed_trajectory local_fast2d_plugin)
    ament_target_dependencies(test_timed_trajectory tf2 tf2_geometry_msgs)
  endif()
  ament_add_gtest(test_candidate_risk_scorer test/test_candidate_risk_scorer.cpp)
  if(TARGET test_candidate_risk_scorer)
    target_link_libraries(test_candidate_risk_scorer local_fast2d_plugin)
    ament_target_dependencies(test_candidate_risk_scorer m1_scope_msgs m1_scope_risk)
  endif()
  ament_add_gtest(test_candidate_selector test/test_candidate_selector.cpp)
  if(TARGET test_candidate_selector)
    target_link_libraries(test_candidate_selector local_fast2d_plugin)
  endif()
  ament_add_gtest(test_restricted_mppi_reference test/test_restricted_mppi_reference.cpp)
  if(TARGET test_restricted_mppi_reference)
    target_link_libraries(test_restricted_mppi_reference local_fast2d_plugin)
  endif()
  ament_add_gtest(test_restricted_mppi_policy test/test_restricted_mppi_policy.cpp)
  if(TARGET test_restricted_mppi_policy)
    target_link_libraries(test_restricted_mppi_policy local_fast2d_plugin)
  endif()
  ament_add_gtest(test_restricted_mppi_phase_tracker test/test_restricted_mppi_phase_tracker.cpp)
  if(TARGET test_restricted_mppi_phase_tracker)
    target_link_libraries(test_restricted_mppi_phase_tracker local_fast2d_plugin)
  endif()
  ament_add_gtest(test_timed_reference_retimer test/test_timed_reference_retimer.cpp)
  if(TARGET test_timed_reference_retimer)
    target_link_libraries(test_timed_reference_retimer local_fast2d_plugin)
  endif()
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
