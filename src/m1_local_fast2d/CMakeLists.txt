cmake_minimum_required(VERSION 3.8)
project(m1_local_fast2d)
find_package(ament_cmake REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(m1_fast_planner REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/hybrid_controller.cpp src/local_fast2d_core.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} diagnostic_msgs m1_fast_planner nav2_core nav2_costmap_2d nav2_mppi_controller nav2_msgs nav_msgs pluginlib rclcpp std_msgs tf2 tf2_geometry_msgs)
add_executable(m1_local_fast2d_causality_harness src/causality_harness.cpp)
target_link_libraries(m1_local_fast2d_causality_harness ${PROJECT_NAME})
ament_target_dependencies(m1_local_fast2d_causality_harness diagnostic_msgs nav2_msgs nav_msgs rclcpp std_msgs tf2)
pluginlib_export_plugin_description_file(nav2_core hybrid_controller_plugin.xml)
install(TARGETS ${PROJECT_NAME} m1_local_fast2d_causality_harness LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES hybrid_controller_plugin.xml DESTINATION share/${PROJECT_NAME})
ament_package()
